# Salin seluruh isi file ini ke .github/workflows/build-apk.yml di GitHub (lihat README langkah 3)
name: Build APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Ambil kode
        uses: actions/checkout@v4

      - name: Pasang Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Pasang Java 21
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21

      - name: Pasang dependensi
        run: npm install

      - name: Buat proyek Android
        run: npx cap add android

      - name: Pasang ikon aplikasi
        run: cp -r icons/res/. android/app/src/main/res/

      - name: Sinkronkan halaman web ke Android
        run: npx cap sync android

      - name: Bangun APK
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon

      - name: Unggah APK
        uses: actions/upload-artifact@v4
        with:
          name: Aplikasi-VDOT-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
